package com.example.inventorytracker;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteConstraintException;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.util.ArrayList;
import java.util.List;

/** Owns the persistent account and inventory tables. */
public final class InventoryDatabaseHelper extends SQLiteOpenHelper {

    private static final String DATABASE_NAME = "inventory_tracker.db";
    private static final int DATABASE_VERSION = 1;

    private static final String TABLE_USERS = "users";
    private static final String TABLE_INVENTORY = "inventory";

    public InventoryDatabaseHelper(Context context) {
        super(context, DATABASE_NAME, null, DATABASE_VERSION);
    }

    @Override
    public void onCreate(SQLiteDatabase database) {
        database.execSQL(
                "CREATE TABLE " + TABLE_USERS + " (" +
                        "id INTEGER PRIMARY KEY AUTOINCREMENT, " +
                        "username TEXT NOT NULL UNIQUE COLLATE NOCASE, " +
                        "password_hash TEXT NOT NULL, " +
                        "password_salt TEXT NOT NULL)"
        );
        database.execSQL(
                "CREATE TABLE " + TABLE_INVENTORY + " (" +
                        "id INTEGER PRIMARY KEY AUTOINCREMENT, " +
                        "name TEXT NOT NULL, " +
                        "sku TEXT NOT NULL UNIQUE COLLATE NOCASE, " +
                        "quantity INTEGER NOT NULL CHECK(quantity >= 0), " +
                        "low_stock_threshold INTEGER NOT NULL CHECK(low_stock_threshold >= 0))"
        );
    }

    @Override
    public void onUpgrade(SQLiteDatabase database, int oldVersion, int newVersion) {
        // Version 1 is the initial schema; future migrations belong here.
    }

    public boolean createUser(String username, String password) {
        String salt = PasswordHasher.newSalt();
        ContentValues values = new ContentValues();
        values.put("username", username);
        values.put("password_hash", PasswordHasher.hash(password, salt));
        values.put("password_salt", salt);
        try {
            return getWritableDatabase().insertOrThrow(TABLE_USERS, null, values) != -1;
        } catch (SQLiteConstraintException exception) {
            return false;
        }
    }

    public boolean authenticate(String username, String password) {
        try (Cursor cursor = getReadableDatabase().query(
                TABLE_USERS,
                new String[]{"password_hash", "password_salt"},
                "username = ?",
                new String[]{username},
                null,
                null,
                null,
                "1"
        )) {
            if (!cursor.moveToFirst()) {
                return false;
            }
            String expectedHash = cursor.getString(cursor.getColumnIndexOrThrow("password_hash"));
            String salt = cursor.getString(cursor.getColumnIndexOrThrow("password_salt"));
            String actualHash = PasswordHasher.hash(password, salt);
            return MessageDigest.isEqual(
                    expectedHash.getBytes(StandardCharsets.UTF_8),
                    actualHash.getBytes(StandardCharsets.UTF_8)
            );
        }
    }

    public boolean addItem(String name, String sku, int quantity, int threshold) {
        ContentValues values = new ContentValues();
        values.put("name", name);
        values.put("sku", sku);
        values.put("quantity", quantity);
        values.put("low_stock_threshold", threshold);
        try {
            return getWritableDatabase().insertOrThrow(TABLE_INVENTORY, null, values) != -1;
        } catch (SQLiteConstraintException exception) {
            return false;
        }
    }

    public void updateItemQuantity(long itemId, int quantity) {
        ContentValues values = new ContentValues();
        values.put("quantity", quantity);
        getWritableDatabase().update(
                TABLE_INVENTORY,
                values,
                "id = ?",
                new String[]{String.valueOf(itemId)}
        );
    }

    public void deleteItem(long itemId) {
        getWritableDatabase().delete(
                TABLE_INVENTORY,
                "id = ?",
                new String[]{String.valueOf(itemId)}
        );
    }

    public List<InventoryItem> getAllItems() {
        return queryItems(null, null, "name COLLATE NOCASE ASC");
    }

    public List<InventoryItem> getLowStockItems() {
        return queryItems("quantity <= low_stock_threshold", null, "quantity ASC");
    }

    private List<InventoryItem> queryItems(
            String selection,
            String[] selectionArguments,
            String orderBy
    ) {
        List<InventoryItem> items = new ArrayList<>();
        try (Cursor cursor = getReadableDatabase().query(
                TABLE_INVENTORY,
                new String[]{"id", "name", "sku", "quantity", "low_stock_threshold"},
                selection,
                selectionArguments,
                null,
                null,
                orderBy
        )) {
            while (cursor.moveToNext()) {
                items.add(new InventoryItem(
                        cursor.getLong(cursor.getColumnIndexOrThrow("id")),
                        cursor.getString(cursor.getColumnIndexOrThrow("name")),
                        cursor.getString(cursor.getColumnIndexOrThrow("sku")),
                        cursor.getInt(cursor.getColumnIndexOrThrow("quantity")),
                        cursor.getInt(cursor.getColumnIndexOrThrow("low_stock_threshold"))
                ));
            }
        }
        return items;
    }
}

package com.example.inventorytracker;

import android.Manifest;
import android.content.pm.PackageManager;

import java.util.List;

/** Coordinates the login, inventory, item-entry, and notification screens. */
public class MainActivity extends AppCompatActivity {

    private static final int SMS_PERMISSION_REQUEST = 1001;

    private InventoryDatabaseHelper database;
    private String pendingSmsNumber;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        database = new InventoryDatabaseHelper(this);
        showLogin(null);

    @Override
    protected void onDestroy() {
        database.close();

    }

    public void showLogin(View view) {
        setContentView(R.layout.activity_main);
        findViewById(R.id.signInButton).setOnClickListener(ignored -> signIn());
    }

    public void showCreateAccount(View view) {
        setContentView(R.layout.activity_create_account);
        findViewById(R.id.createAccountButton).setOnClickListener(ignored -> createAccount());
    }

    public void showInventory(View view) {
        setContentView(R.layout.activity_inventory);
        renderInventory();
    }

    public void showAddItem(View view) {
        setContentView(R.layout.activity_add_item);
        findViewById(R.id.saveItemButton).setOnClickListener(ignored -> saveItem());
    }

    public void showNotifications(View view) {
        setContentView(R.layout.activity_notifications);
        updateSmsPermissionStatus();
    }

    private void signIn() {
        String username = textOf(R.id.usernameInput);
        String password = textOf(R.id.passwordInput);

        if (username.isEmpty() || password.isEmpty()) {
            toast(R.string.enter_credentials);
            return;
        }
        if (!database.authenticate(username, password)) {
            toast(R.string.invalid_credentials);
            return;
        }
        showInventory(null);
    }

    private void createAccount() {
        String username = textOf(R.id.createUsernameInput);
        String password = textOf(R.id.createPasswordInput);
        String confirmation = textOf(R.id.confirmPasswordInput);

        if (username.isEmpty() || password.isEmpty()) {
            toast(R.string.enter_credentials);
            return;
        }
        if (!password.equals(confirmation)) {
            toast(R.string.passwords_do_not_match);
            return;
        }
        if (!database.createUser(username, password)) {
            toast(R.string.username_exists);
            return;
        }

        toast(R.string.account_created);
        showInventory(null);
    }

    private void saveItem() {
        String name = textOf(R.id.itemNameInput);
        String sku = textOf(R.id.skuInput);
        Integer quantity = integerOf(R.id.quantityInput);
        Integer threshold = integerOf(R.id.thresholdInput);

        if (name.isEmpty() || sku.isEmpty() || quantity == null || threshold == null) {
            toast(R.string.complete_item_fields);
            return;
        }
        if (!database.addItem(name, sku, quantity, threshold)) {
            toast(R.string.sku_exists);
            return;
        }

        toast(R.string.item_saved);
        showInventory(null);
    }

    private void renderInventory() {
        List<InventoryItem> items = database.getAllItems();
        TextView subtitle = findViewById(R.id.inventorySubtitle);
        subtitle.setText(getResources().getQuantityString(
                R.plurals.inventory_count,
                items.size(),
                items.size(),
                countLowStock(items)

        TableLayout table = findViewById(R.id.inventoryTable);
        TextView emptyState = findViewById(R.id.emptyInventoryMessage);
        emptyState.setVisibility(items.isEmpty() ? View.VISIBLE : View.GONE);

        for (InventoryItem item : items) {
            table.addView(createInventoryRow(item));
        }

    private TableRow createInventoryRow(InventoryItem item) {
        TableRow row = new TableRow(this);
        row.setGravity(Gravity.CENTER_VERTICAL);
        row.setPadding(0, dp(8), 0, dp(8));

        LinearLayout nameCell = new LinearLayout(this);
        nameCell.setOrientation(LinearLayout.VERTICAL);
        TextView name = bodyText(item.getName());
        name.setTextColor(Color.rgb(23, 32, 51));
        TextView sku = bodyText(item.getSku());
        sku.setTextSize(12);
        sku.setTextColor(Color.rgb(95, 104, 120));
        nameCell.addView(name);
        nameCell.addView(sku);
        row.addView(nameCell, new TableRow.LayoutParams(dp(108), TableRow.LayoutParams.WRAP_CONTENT));

        quantityCell.setGravity(Gravity.CENTER);
        Button decrement = compactButton("−", 28);

        quantity.setGravity(Gravity.CENTER);
        quantity.setMinWidth(dp(32));
        quantity.setPadding(0, 0, 0, 0);
        Button increment = compactButton("+", 28);
        decrement.setOnClickListener(ignored -> updateQuantity(item, Math.max(0, item.getQuantity() - 1)));
        increment.setOnClickListener(ignored -> updateQuantity(item, item.getQuantity() + 1));

        quantityCell.addView(quantity);
        quantityCell.addView(increment);

    private void updateQuantity(InventoryItem item, int quantity) {
        database.updateItemQuantity(item.getId(), quantity);
        showInventory(null);
    }

    public void requestSmsPermission(View view) {
        String phoneNumber = textOf(R.id.phoneNumberInput);
        if (phoneNumber.isEmpty()) {
            toast(R.string.enter_phone_number);
            return;
        }
        pendingSmsNumber = phoneNumber;

        if (ContextCompat.checkSelfPermission(this, Manifest.permission.SEND_SMS)
                == PackageManager.PERMISSION_GRANTED) {
            sendLowStockAlert();
            return;
        }

        ActivityCompat.requestPermissions(
                this,
                new String[]{Manifest.permission.SEND_SMS},
                SMS_PERMISSION_REQUEST
        );
    }

    @Override
    public void onRequestPermissionsResult(
            int requestCode,
            @NonNull String[] permissions,
            @NonNull int[] grantResults
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults);
        if (requestCode != SMS_PERMISSION_REQUEST) {
            return;
        }

        updateSmsPermissionStatus();
        if (grantResults.length > 0 && grantResults[0] == PackageManager.PERMISSION_GRANTED) {
            sendLowStockAlert();
        } else {
            toast(R.string.sms_denied_inventory_works);
        }
    }

    @SuppressWarnings("deprecation")
    private void sendLowStockAlert() {
        List<InventoryItem> lowStockItems = database.getLowStockItems();
        if (lowStockItems.isEmpty()) {
            toast(R.string.no_low_stock_items);
            return;
        }

        InventoryItem item = lowStockItems.get(0);
        String message = getString(
                R.string.low_stock_message,
                item.getName(),
                item.getQuantity(),
                item.getLowStockThreshold()
        );
        try {
            SmsManager.getDefault().sendTextMessage(pendingSmsNumber, null, message, null, null);
            toast(R.string.sms_sent);
        } catch (RuntimeException exception) {
            toast(R.string.sms_unavailable);
        }
    }

    private void updateSmsPermissionStatus() {
        TextView status = findViewById(R.id.smsPermissionStatus);
        if (status == null) {
            return;
        }
        boolean granted = ContextCompat.checkSelfPermission(this, Manifest.permission.SEND_SMS)
                == PackageManager.PERMISSION_GRANTED;
        status.setText(granted ? R.string.sms_permission_granted : R.string.sms_permission_not_granted);
        status.setTextColor(granted ? Color.rgb(25, 113, 60) : Color.rgb(95, 104, 120));
    }

    private int countLowStock(List<InventoryItem> items) {
        int count = 0;
        for (InventoryItem item : items) {
            if (item.isLowStock()) {
                count++;
            }
        }
        return count;
    }

    private String textOf(int viewId) {
        EditText input = findViewById(viewId);
        return input.getText().toString().trim();
    }

    private Integer integerOf(int viewId) {
        try {
            int value = Integer.parseInt(textOf(viewId));
            return value < 0 ? null : value;
        } catch (NumberFormatException exception) {
            return null;
        }
    }

    private TextView bodyText(String value) {
        TextView view = new TextView(this);
        view.setText(value);
        view.setTextSize(14);
        view.setPadding(dp(4), dp(4), dp(4), dp(4));
        return view;
    }

    private Button compactButton(String label, int widthDp) {
        Button button = new Button(this);
        button.setText(label);
        button.setTextSize(11);
        button.setAllCaps(false);
        button.setWidth(dp(widthDp));
        button.setMinWidth(0);
        button.setMinimumWidth(0);
        button.setMinHeight(dp(40));
        button.setPadding(0, 0, 0, 0);
        return button;
    }

    private int dp(int value) {
        return Math.round(value * getResources().getDisplayMetrics().density);
    }

    private void toast(int messageId) {
        Toast.makeText(this, messageId, Toast.LENGTH_SHORT).show();
    }
}

# CS-360 Project Three: Inventory Tracker

This Android Studio project is the finished version of my inventory app. It uses Java, XML layouts, and a local SQLite database. The minimum supported Android version is API 33.

## Functional flow

- A new user can create an account, and returning users can sign in with the saved credentials. Passwords are salted and hashed before they are stored.
- The inventory screen reads saved items from SQLite and presents them in a grid. A user can add an item, increase or decrease its quantity, or delete it. The data remains available after the app closes and reopens.
- Each item has a low-stock threshold. The inventory summary and status column show which items need attention.
- The notifications screen requests SMS permission only when the user tries to send an alert. If permission is denied, the inventory functions continue to work. If permission is granted, the app can send a text alert for a low-stock item.
- The screens provide a complete path through sign-in, account creation, inventory, item entry, notifications, and sign-out (OpenAI, 2026).

package com.example.inventorytracker;

import android.util.Base64;

import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.security.spec.InvalidKeySpecException;

import javax.crypto.SecretKeyFactory;
import javax.crypto.spec.PBEKeySpec;

/** Hashes account passwords before they are stored in SQLite. */
final class PasswordHasher {

    private static final int ITERATIONS = 120_000;
    private static final int KEY_LENGTH = 256;
    private static final int SALT_BYTES = 16;

    private PasswordHasher() {
    }

    static String newSalt() {
        byte[] salt = new byte[SALT_BYTES];
        new SecureRandom().nextBytes(salt);
        return Base64.encodeToString(salt, Base64.NO_WRAP);
    }

    static String hash(String password, String encodedSalt) {
        byte[] salt = Base64.decode(encodedSalt, Base64.NO_WRAP);
        PBEKeySpec specification = new PBEKeySpec(
                password.toCharArray(),
                salt,
                ITERATIONS,
                KEY_LENGTH
        );
        try {
            SecretKeyFactory factory = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256");
            return Base64.encodeToString(factory.generateSecret(specification).getEncoded(), Base64.NO_WRAP);
        } catch (NoSuchAlgorithmException | InvalidKeySpecException exception) {
            throw new IllegalStateException("Unable to hash the password", exception);
        } finally {
            specification.clearPassword();
        }
    }
}

        db.beginTransaction();

            InventoryItem current = getItem(preview.item.getId());
            if (current == null) return AdjustmentResult.MISSING;
            if (current.getQuantity() != preview.item.getQuantity()) return AdjustmentResult.STALE;
            StockAdjustment checked = new StockAdjustment(current, preview.delta, preview.reason);
            ContentValues values = new ContentValues();
            values.put("quantity", checked.quantity);
            int updated = db.update(TABLE_INVENTORY, values, "id = ? AND quantity = ?",

            if (updated != 1) throw new IllegalStateException("Adjustment did not update exactly one item.");
            ContentValues movement = new ContentValues();

package com.example.inventorytracker;

import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.PriorityQueue;

/** Selects from repository records with unique IDs and nonnegative int stock values. */
public final class RestockSelector {
    private RestockSelector() { }

    public static final Comparator<InventoryItem> PRIORITY = (a, b) -> {
        int stockout = Boolean.compare(b.getQuantity() == 0, a.getQuantity() == 0);
        if (stockout != 0) return stockout;
        if (a.getQuantity() != 0) {
            long left = (long) a.getQuantity() * b.getLowStockThreshold();
            long right = (long) b.getQuantity() * a.getLowStockThreshold();
            int coverage = Long.compare(left, right);
            if (coverage != 0) return coverage;
        }
        int sku = a.getSku().compareTo(b.getSku());
        return sku != 0 ? sku : Long.compare(a.getId(), b.getId());
    };

    /** For k > 0 and h = min(k, eligible), scans n records in O(n + eligible * log(h+1))
     * time, then orders h results in O(h * log(h+1)); O(h) storage favors short lists. */
    public static List<InventoryItem> select(List<InventoryItem> items, int k) {
        if (k < 0) throw new IllegalArgumentException("Enter zero or a positive whole number.");
        if (k == 0) return new ArrayList<>();
        // Reversing the output order puts the least urgent retained item at the root.
        PriorityQueue<InventoryItem> heap = new PriorityQueue<>(PRIORITY.reversed());
        for (InventoryItem item : items) {
            if (item.getQuantity() < 0 || item.getLowStockThreshold() < 0)
                throw new IllegalArgumentException("Stock values cannot be negative.");
            if (!item.isLowStock()) continue;
            if (heap.size() < k) heap.add(item);
            else if (PRIORITY.compare(item, heap.peek()) < 0) {
                heap.remove();
                heap.add(item);
            }
        }
        List<InventoryItem> result = new ArrayList<>(heap);
        result.sort(PRIORITY);
        return result;
    }

    public static List<InventoryItem> fullSort(List<InventoryItem> items, int k) {
        if (k < 0) throw new IllegalArgumentException("Enter zero or a positive whole number.");
        List<InventoryItem> eligible = new ArrayList<>();
        if (k == 0) return eligible;
        for (InventoryItem item : items) {
            if (item.getQuantity() < 0 || item.getLowStockThreshold() < 0)
                throw new IllegalArgumentException("Stock values cannot be negative.");
            if (item.isLowStock()) eligible.add(item);
        }
        eligible.sort(PRIORITY);
        return new ArrayList<>(eligible.subList(0, Math.min(k, eligible.size())));
    }
}

## Code responsibilities

- `MainActivity`: screens, input, permission handling, main-thread rendering.
- `InventoryViewModel`: retained screen/operation state and a serial background executor for account hashing and all database work. Its `Phase` transitions are EDITING → LOADING → PREVIEW → SAVING → SUCCESS or ERROR.

- `InventoryItem` and `PasswordHasher`: unchanged baseline data and hashing logic.

## Selection complexity and benchmark

After loading n items with m eligible candidates, let h = min(k,m). For k > 0, the heap uses O(h) auxiliary storage, O(n + m log(h+1)) selection time, and O(h log(h+1)) output sorting. The full-sort reference uses O(m) storage and O(n + m log(m+1)) time. Both receive the existing O(n) repository list; the SQL name ordering, loading and UI rendering are excluded from these bounds and timings.

package com.example.inventorytracker;

import android.util.Base64;

import java.security.NoSuchAlgorithmException;
import java.security.SecureRandom;
import java.security.spec.InvalidKeySpecException;

import javax.crypto.SecretKeyFactory;
import javax.crypto.spec.PBEKeySpec;

/** Hashes account passwords before they are stored in SQLite. */
final class PasswordHasher {

    private static final int ITERATIONS = 120_000;
    private static final int KEY_LENGTH = 256;
    private static final int SALT_BYTES = 16;

    private PasswordHasher() {
    }

    static String newSalt() {
        byte[] salt = new byte[SALT_BYTES];
        new SecureRandom().nextBytes(salt);
        return Base64.encodeToString(salt, Base64.NO_WRAP);
    }

    static String hash(String password, String encodedSalt) {
        byte[] salt = Base64.decode(encodedSalt, Base64.NO_WRAP);
        PBEKeySpec specification = new PBEKeySpec(
                password.toCharArray(),
                salt,
                ITERATIONS,
                KEY_LENGTH
        );
        try {
            SecretKeyFactory factory = SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256");
            return Base64.encodeToString(factory.generateSecret(specification).getEncoded(), Base64.NO_WRAP);
        } catch (NoSuchAlgorithmException | InvalidKeySpecException exception) {
            throw new IllegalStateException("Unable to hash the password", exception);
        } finally {
            specification.clearPassword();
        }
    }
}

package com.example.inventorytracker;

import org.junit.Test;
import static org.junit.Assert.*;
import java.util.*;
import java.math.BigInteger;

public class RestockSelectorTest {
    private InventoryItem item(long id, String sku, int q, int t) {
        return new InventoryItem(id, sku, sku, q, t);
    }
    private List<Long> ids(List<InventoryItem> items) {
        List<Long> result = new ArrayList<>();
        for (InventoryItem i : items) result.add(i.getId());
        return result;
    }
    @Test public void ranksStockoutsBeforeRelativeCoverage() {
        List<InventoryItem> input = Arrays.asList(item(1,"C",2,3), item(2,"D",3,30),
                item(3,"B",0,0), item(4,"A",0,20), item(5,"E",7,2));
        assertEquals(Arrays.asList(4L,3L,2L),ids(RestockSelector.select(input,3)));
        assertEquals(Arrays.asList(4L,3L,2L,1L),ids(RestockSelector.select(input,99)));
    }
    @Test public void equalCoverageUsesSkuThenId() {
        List<InventoryItem> input=Arrays.asList(item(4,"A",2,6),item(3,"A",1,3),item(1,"B",3,9));
        assertEquals(Arrays.asList(3L,4L,1L),ids(RestockSelector.select(input,3)));
    }
    @Test public void boundarySizesDoNotMutateInput() {
        List<InventoryItem> input=Collections.unmodifiableList(Arrays.asList(item(1,"A",2,3)));
        assertTrue(RestockSelector.select(input,0).isEmpty());
        assertTrue(RestockSelector.select(Collections.emptyList(),5).isEmpty());
        assertEquals(Arrays.asList(1L),ids(RestockSelector.select(input,Integer.MAX_VALUE)));
        assertEquals(1,input.size());
    }
    @Test public void rejectsInvalidLimitsAndStock() {
        assertThrows(IllegalArgumentException.class,()->RestockSelector.select(Collections.emptyList(),-1));
        assertThrows(IllegalArgumentException.class,()->RestockSelector.select(Arrays.asList(item(1,"A",-1,2)),1));
        assertThrows(IllegalArgumentException.class,()->RestockSelector.select(Arrays.asList(item(1,"A",0,-1)),1));
    }
    @Test public void maximumIntProductsRemainExact() {
        int m=Integer.MAX_VALUE;
        List<InventoryItem> input=Arrays.asList(item(1,"A",m-1,m),item(2,"B",m-2,m-1));
        assertEquals(Arrays.asList(2L,1L),ids(RestockSelector.select(input,2)));
    }
    @Test public void agreesWithIndependentRationalOracleAcrossRandomInputs() {
        Random r=new Random(499);
        Comparator<InventoryItem> oracle=(a,b)->{
            int zero=Boolean.compare(b.getQuantity()==0,a.getQuantity()==0);
            if(zero!=0)return zero;
            if(a.getQuantity()!=0){
                int ratio=BigInteger.valueOf(a.getQuantity()).multiply(BigInteger.valueOf(b.getLowStockThreshold()))
                    .compareTo(BigInteger.valueOf(b.getQuantity()).multiply(BigInteger.valueOf(a.getLowStockThreshold())));
                if(ratio!=0)return ratio;
            }
            int sku=a.getSku().compareTo(b.getSku());
            return sku!=0?sku:Long.compare(a.getId(),b.getId());
        };
        for(int trial=0;trial<100;trial++) {
            List<InventoryItem> input=new ArrayList<>();
            for(int i=0;i<200;i++)input.add(item(i,"S"+r.nextInt(20),r.nextInt(31),r.nextInt(31)));
            List<InventoryItem> expected=new ArrayList<>();
            for(InventoryItem i:input)if(i.getQuantity()<=i.getLowStockThreshold())expected.add(i);
            expected.sort(oracle);
            Collections.shuffle(input,r);
            for(int k:new int[]{0,1,5,20,199,250}) {
                List<InventoryItem> got=RestockSelector.select(input,k);
                assertEquals(ids(expected.subList(0,Math.min(k,expected.size()))),ids(got));
                assertEquals(got.size(),new HashSet<>(ids(got)).size());
            }
        }
    }
}
